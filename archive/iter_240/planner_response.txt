    L = grid.shape[0]
    total = int(grid.sum())
    if total == 0:
        return None, 0
    coords = np.zeros(3)
    pos = np.arange(L)
    theta = 2 * np.pi * pos / L
    cos_t = np.cos(theta)
    sin_t = np.sin(theta)
    for axis in range(3):
        if axis == 0:
            w = grid.sum(axis=(1, 2, 3)).astype(np.float64)
