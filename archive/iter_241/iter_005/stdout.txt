def detect_period(shapes_list, L):
    n = len(shapes_list)
    for p in range(1, n):
        ok = True
        for t in range(n - p):
            if not is_translate(shapes_list[t], shapes_list[t + p], L):
                ok = False
                break
        if ok:

    return None
