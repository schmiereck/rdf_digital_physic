def _raw_com_and_bits(g):
    """Return the raw (toroidal) CoM and bit count for a grid."""
    rows, cols = np.where(g > 0)
    if len(rows) == 0:
        return (0.0, 0.0), 0
    return (float(np.mean(rows)), float(np.mean(cols))), int(g.sum())
