import math
from typing import Any
import numpy as np

class DisplacementConsistencyFitness:
    """Fitness function that rewards consistent, directional motion.

    Scoring pipeline:
      1. Divide sim_history into num_windows equal-duration windows.
      2. For each window, compute the displacement vector (ΔCOM) and its
         magnitude (velocity in cells/step, normalized by window duration).
      3. Compute the mean of the per-window velocity *vectors* — this gives
         the net directional drift.
      4. Compute the standard deviation of the per-window velocity *magnitudes*
         — this measures consistency of motion speed.
      5. Core formula: base_fitness = mean_velocity_magnitude / (1 + std_dev)
      6. Multiply by a "leaky" bit-conservation score.
      7. final_fitness = base_fitness * total_conservation_score
    """

    name = "DisplacementConsistencyFitness"

    def __init__(
        self,
        num_windows: int = 5,
        bits_per_cell: int = 1,
        strict_conservation: bool = False,
        max_bit_threshold: int | None = None,
        max_velocity_threshold: float | None = None,
    ) -> None:
        self.num_windows = max(2, int(num_windows))
        self.bits_per_cell = int(bits_per_cell)
        self.strict_conservation = strict_conservation
        self.max_bit_threshold = max_bit_threshold
        self.max_velocity_threshold = max_velocity_threshold

    def __call__(
        self,
        sim_history: list[dict[str, Any]],
    ) -> float:
        if not sim_history:
            return 0.0

        sorted_history = sorted(sim_history, key=lambda e: e["step"])

        # — hard gate: max bit threshold —
        if self.max_bit_threshold is not None:
            if any(e["bit_count"] > self.max_bit_threshold for e in sorted_history):
                return 0.0

        # — hard gate: strict conservation —
        if self.strict_conservation:
            initial_bits = sorted_history[0]["bit_count"]
            if any(e["bit_count"] != initial_bits for e in sorted_history):
                return 0.0

        initial_step = float(sorted_history[0]["step"])
        final_step = float(sorted_history[-1]["step"])
        total_steps = final_step - initial_step
        if total_steps <= 0:
            return 0.0

        # — Step 2: per-window velocity vectors —
        steps_per_window = total_steps / self.num_windows
        window_velocity_mags: list[float] = []
        window_velocity_vectors: list[tuple[float, float]] = []

        for w in range(self.num_windows):
            window_start = initial_step + w * steps_per_window
            window_end = window_start + steps_per_window
            first_entry: dict | None = None
            last_entry: dict | None = None

            for entry in sorted_history:
                s = float(entry["step"])
                if s < window_start:
                    continue
                effective_window_end = window_end
                if w == self.num_windows - 1:
                    effective_window_end = final_step
                if s > effective_window_end:
                    break
                if first_entry is None:
                    first_entry = entry
                last_entry = entry

            if first_entry is None or last_entry is None:
                window_velocity_mags.append(0.0)
                window_velocity_vectors.append((0.0, 0.0))
                continue

            dx = last_entry["com"][0] - first_entry["com"][0]
            dy = last_entry["com"][1] - first_entry["com"][1]
            velocity_mag = math.sqrt(dx * dx + dy * dy)
            window_steps = last_entry["step"] - first_entry["step"]
            if window_steps > 0:
                velocity_mag = velocity_mag / window_steps
                dx = dx / window_steps
                dy = dy / window_steps
            else:
                velocity_mag = 0.0
                dx = 0.0
                dy = 0.0

            window_velocity_mags.append(velocity_mag)
            window_velocity_vectors.append((dx, dy))

        # — Step 3: mean velocity vector —
        mean_dx = sum(v[0] for v in window_velocity_vectors) / self.num_windows
        mean_dy = sum(v[1] for v in window_velocity_vectors) / self.num_windows
        mean_velocity_magnitude = math.sqrt(mean_dx * mean_dx + mean_dy * mean_dy)

        # — hard gate: max velocity threshold —
        if self.max_velocity_threshold is not None and mean_velocity_magnitude >= self.max_velocity_threshold:
            return 0.0

        # — Step 4: std dev of velocity magnitudes —
        velocity_magnitudes = np.array(window_velocity_mags, dtype=np.float64)
        std_dev_velocity_magnitudes = float(np.std(velocity_magnitudes))

        # — Step 5: core fitness formula —
        if mean_velocity_magnitude == 0.0:
            base_fitness = 0.0
        else:
            base_fitness = mean_velocity_magnitude / (1.0 + std_dev_velocity_magnitudes)

        # — Step 6: leaky conservation score —
        total_conservation_score = self._compute_conservation_score(sorted_history)

        # — Step 7: final score —
        fitness = base_fitness * total_conservation_score
        return float(fitness)

    def _compute_conservation_score(
        self, sim_history: list[dict[str, Any]]
    ) -> float:
        """Leaky bit-conservation score in range [0, 1]."""
        if not sim_history:
            return 1.0
        initial_bits = sim_history[0]["bit_count"]
        if initial_bits == 0:
            return 1.0
        factors: list[float] = []
        for entry in sim_history:
            bc = entry["bit_count"]
            if bc == initial_bits:
                factors.append(1.0)
            else:
                factors.append(min(bc, initial_bits) / max(bc, initial_bits))
        return float(sum(factors) / len(factors))

## Analysis: Why it Failed to Reject the Breeder

### The core problem

The fitness function is designed to detect **drifters** — objects whose velocity vectors change direction unpredictably across windows. It does this by:

1. Computing per-window **COM displacement** vectors
2. Computing the **mean velocity magnitude** (numerator) and **std dev of velocity magnitudes** (denominator)
3. Scoring as `mean_mag / (1 + std_dev)` × conservation_score

### Why the breeder slips through

The breeder (the rule being rejected) likely does **not** behave like a drifter. It behaves like a **true multi-particle system**: a glider that spawns and leaves behind stationary/walking debris. In this scenario:

| Aspect | Breeder | Drifter | What fitness measures |
|---|---|---|---|
| COM per window | **Consistent** direction & speed | Unpredictable | Low std dev → **good score** |
| Bit count | Changes (debris created) | Stable | Conservation penalty |
| Cohesion | Multiple disconnected objects | Single oscillating object | **Not measured at all** |

**The fitness function has no concept of "cohesion" or "single unit".** It only looks at COM displacement. If the breeder's overall COM moves in a consistent direction at consistent speed, it will get a high consistency score — even though it's fundamentally not a glider.

### Key gaps to fix

1. **No bounding-box / spread measurement** — The fitness does not check whether all live cells stay within a bounded region as they move. A true glider has a small, bounded bounding box. A breeder with debris has an ever-growing bounding box.

2. **No object-count / component analysis** — It does not check whether the live cells form a single connected component. A breeder produces multiple disconnected objects.

3. **No per-window bit-count tracking** — The bit count is only used for conservation scoring, but a growing bit count (debris creation) over windows is a hallmark of breeders.

4. **No initial-cell vacuum check** — The fitness never verifies that the initial seed cells are empty by the end of the simulation. A breeder pattern would still have cells at the starting positions (or their debris would occupy them).

### How to improve it

Add these gates/metrics to the fitness function:

1. **Bounding box growth constraint** — At each step, compute the bounding box of all live cells. The bounding box area must stay below a threshold (or at least not grow monotonically). A glider's bounding box is bounded; a breeder's grows as debris accumulates.

2. **Connected-component count** — Use a flood-fill / connected-components algorithm on the grid. A true glider has exactly 1 component; a breeder with debris has ≥2.

3. **Initial-cell vacuum enforcement** — Check that the final step has **zero cells** at or near the original seed positions. This ensures the pattern truly "moves" rather than just expanding from a static anchor.

4. **Per-window bit-count trend** — Beyond conservation, check that the bit count is stable *across windows* (not just globally). Monotonic growth across windows = breeder.
